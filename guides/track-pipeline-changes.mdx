description: "Poll the changes feed for a watchlist of companies."

Turn a watchlist into company ids and read every change since the last run.
Requires `PAV_API_KEY` and `pip install requests`.

1. `GET /v1/search?types=company` returns each company's id.
2. `GET /v1/changes?company_id=<ids>&since=<checkpoint>` lists changes,
   newest first; follow `next_cursor` to the end.
3. Save the newest `detected_at` as the next checkpoint. `since` is
   inclusive, so skip ids already seen.

```python track_changes.py
import json
import os
import pathlib

import requests

API = "https://api.pav.bio/v1"
HEADERS = {"Authorization": f"Bearer {os.environ['PAV_API_KEY']}"}
STATE = pathlib.Path("pav_changes_state.json")


def get(path, **params):
    resp = requests.get(f"{API}{path}", headers=HEADERS, params=params, timeout=60)
    resp.raise_for_status()
    return resp.json()


ids = [get("/search", q=name, types="company", limit=1)["data"][0]["id"]
       for name in ["Eli Lilly", "Novo Nordisk", "Pfizer"]]
state = json.loads(STATE.read_text()) if STATE.exists() else {"since": "2026-08-01T00:00:00Z", "seen": []}

events, params = [], {"company_id": ",".join(ids), "since": state["since"], "limit": 200}
while True:
    page = get("/changes", **params)
    events += [e for e in page["data"] if e["id"] not in state["seen"]]
    if "next_cursor" not in page:
        break
    params["cursor"] = page["next_cursor"]

print(len(events), "new changes; latest:")
for e in events[:5]:
    phase = f"{e.get('prev_phase', '')} -> {e.get('new_phase', '')}"
    print(f"{e['detected_at'][:10]} {e['change_type']:<6} {e['company']} | {e['drug']} | {e.get('indication', '')} | {phase}")

if events:
    newest = events[0]["detected_at"]
    STATE.write_text(json.dumps({"since": newest, "seen": [e["id"] for e in events if e["detected_at"] == newest]}))
```
